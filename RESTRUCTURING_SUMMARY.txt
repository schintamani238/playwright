HOW TO USE
═══════════════════════════════════════════════════════════════════════════════

RUN TESTS
  npm test                        # Run all tests
  npm test -- tests/specs/        # Run POM tests only
  npm run test:headed             # Run with browser visible
  npm run test:debug              # Interactive debugging mode
  npm run test:report             # View HTML report

WRITE NEW TESTS
  1. Import page objects
     import { LoginPage } from '../../pages/LoginPage';
  
  2. Initialize in beforeEach
     const loginPage = new LoginPage(page);
  
  3. Use page methods in test
     await loginPage.navigateToLogin();
     await loginPage.loginAsStandardUser();

ADD NEW PAGES
  1. Create pages/NewPage.js
  2. Extend BasePage
  3. Add selectors and methods
  4. Import and use in tests
